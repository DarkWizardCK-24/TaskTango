    // ---- TaskTango (everything lives under apps/tasktango/) --------------------------------
    function tt_owner(uid) {
      return request.auth != null && request.auth.uid == uid;
    }

    function tt_task_ok(d) {
      return d.keys().hasOnly([
               'title', 'notes', 'priority', 'due', 'time', 'listId', 'tags', 'subtasks',
               'completed', 'completedAt', 'createdAt', 'order', 'updatedAt'
             ])
             && d.keys().hasAll([
               'title', 'priority', 'listId', 'completed', 'createdAt', 'order', 'updatedAt'
             ])
             && d.title is string && d.title.size() > 0 && d.title.size() <= 300
             && (!('notes' in d) || (d.notes is string && d.notes.size() <= 5000))
             && d.priority in ['low', 'medium', 'high']
             && (!('due' in d) || d.due == null || (d.due is string && d.due.size() == 10))
             && (!('time' in d) || d.time == null || (d.time is string && d.time.size() == 5))
             && d.listId is string && d.listId.size() > 0 && d.listId.size() <= 80
             && (!('tags' in d) || (d.tags is list && d.tags.size() <= 12))
             && (!('subtasks' in d) || (d.subtasks is list && d.subtasks.size() <= 100))
             && d.completed is bool
             && (!('completedAt' in d) || d.completedAt == null || d.completedAt is number)
             && d.createdAt is number
             && d.order is number
             && d.updatedAt is number;
    }

    function tt_meta_ok(d) {
      return d.keys().hasOnly(['lists', 'sort', 'updatedAt'])
             && d.keys().hasAll(['lists', 'sort', 'updatedAt'])
             && d.lists is list && d.lists.size() <= 50
             && d.sort in ['manual', 'due', 'priority', 'created']
             && d.updatedAt is number;
    }

    match /apps/tasktango/users/{uid}/tasks/{taskId} {
      allow read, delete: if tt_owner(uid);
      allow create, update: if tt_owner(uid) && tt_task_ok(request.resource.data);
    }

    match /apps/tasktango/users/{uid}/meta/state {
      allow read, delete: if tt_owner(uid);
      allow create, update: if tt_owner(uid) && tt_meta_ok(request.resource.data);
    }
